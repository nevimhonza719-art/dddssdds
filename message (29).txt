local autofarm_on = true
local FARM_SPEED = 21
local farm_mode = "Basic"
local DOWN_DEPTH = 14
local DOWN_RISE_XZ = 4

local farm_target = nil
local nc_cache = {}
local round_mod = nil
local collected_ids = {}
local collected_count = 0
local was_down = false
local down_ref_y = nil
local last_touch = 0

local function farm_hrp()
    local c = lp.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function get_round_data()
    if not round_mod then
        local ok, m = pcall(function()
            return require(ReplicatedStorage:WaitForChild("Modules"):WaitForChild("CurrentRoundClient"))
        end)
        if ok and type(m) == "table" then round_mod = m end
    end
    return round_mod and round_mod.PlayerData
end

local function can_farm()
    local char = lp.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    local d = get_round_data()
    if type(d) == "table" then
        local me = d[lp.Name]
        if not me or not me.Role or me.Dead then return false end
    end
    return true
end

local function coin_bags_full()
    local pg = lp:FindFirstChild("PlayerGui")
    local main = pg and pg:FindFirstChild("MainGUI")
    local gg = main and main:FindFirstChild("Game")
    local bags = gg and gg:FindFirstChild("CoinBags")
    local container = bags and bags:FindFirstChild("Container")
    if not container then return false end
    local any = false
    for _, v in ipairs(container:GetChildren()) do
        if v:IsA("Frame") and v.Visible then
            any = true
            local full = v:FindFirstChild("Full")
            if not (full and full.Visible) then
                return false
            end
        end
    end
    return any
end

local function reset_farm_progress()
    collected_ids = {}
    collected_count = 0
    farm_target = nil
end

    end
end)

local function coin_available(v)
    return v and v.Parent and v:IsA("BasePart") and not v:GetAttribute("Collected") and not v:GetAttribute("Delete")
end

local function available_coins()
    local list = {}
    for _, v in ipairs(CollectionService:GetTagged("CoinVisual")) do
        if coin_available(v) then
            list[#list + 1] = v
        end
    end
    if #list == 0 then
        for _, c in ipairs(Workspace:GetDescendants()) do
            if c.Name == "Coin_Server" and coin_available(c) then
                list[#list + 1] = c
            end
        end
    end
    return list
end

local function nearest_coin(pos, list)
    local best, bd = nil, math.huge
    for _, v in ipairs(list) do
        local d = (v.Position - pos).Magnitude
        if d < bd then bd = d; best = v end
    end
    return best
end

local function set_farm_noclip(on)
    local char = lp.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if on then
        if not char then return end
        if hum then pcall(function() hum.PlatformStand = true end) end
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") and p.CanCollide then
                if nc_cache[p] == nil then nc_cache[p] = p.CanCollide end
                p.CanCollide = false
            end
        end
    else
        if hum then pcall(function() hum.PlatformStand = false end) end
        for p, v in pairs(nc_cache) do
            if p and p.Parent then pcall(function() p.CanCollide = v end) end
        end
        nc_cache = {}
    end
end

local up_params = RaycastParams.new()
up_params.FilterType = Enum.RaycastFilterType.Exclude
up_params.IgnoreWater = true

local function return_to_surface()
    local hrp = farm_hrp()
    if not hrp then return end
    local origin = hrp.Position
    up_params.FilterDescendantsInstances = { lp.Character }
    local res = Workspace:Raycast(origin, Vector3.new(0, 400, 0), up_params)
    local y = res and (res.Position.Y + 5) or (down_ref_y and down_ref_y + 5 or nil)
    if not y then return end
    pcall(function()
        hrp.CFrame = CFrame.new(origin.X, y, origin.Z)
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end)
end

local function farm_release()
    if was_down then
        was_down = false
        return_to_surface()
    end
    set_farm_noclip(false)
end

local function flat_dist(a, b)
    return Vector3.new(a.X - b.X, 0, a.Z - b.Z).Magnitude
end

local function touch_targets(coin)
    local list = {}
    local seen = {}
    local function add(p)
        if p and not seen[p] and p:IsA("BasePart") and p:FindFirstChildOfClass("TouchTransmitter") then
            seen[p] = true
            list[#list + 1] = p
        end
    end
    add(coin)
    for _, v in ipairs(coin:GetChildren()) do add(v) end
    local par = coin.Parent
    if par then
        if par:IsA("BasePart") then add(par) end
        for _, v in ipairs(par:GetChildren()) do add(v) end
    end
    if #list == 0 then list[1] = coin end
    return list
end

local function fire_touch(coin)
    if type(firetouchinterest) ~= "function" then return end
    if not coin or not coin.Parent then return end
    local now = os.clock()
    if now - last_touch < 0.05 then return end
    last_touch = now
    local hrp = farm_hrp()
    if not hrp then return end
    for _, p in ipairs(touch_targets(coin)) do
        pcall(firetouchinterest, hrp, p, 0)
        pcall(firetouchinterest, hrp, p, 1)
    end
end

local function farm_move(hrp, dest, dt)
    local dir = dest - hrp.Position
    local dist = dir.Magnitude
    local np = dest
    if dist > 0.1 then
        np = hrp.Position + dir.Unit * math.min(FARM_SPEED * dt, dist)
    end
    local cf = CFrame.new(np)
    if farm_mode == "Down" then
        cf = cf * CFrame.Angles(-math.pi * 0.5, 0, 0)
        was_down = true
    end
    pcall(function()
        hrp.CFrame = cf
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end)
end

local farm_conn = RunService.Stepped:Connect(function(_, dt)
    if not autofarm_on then return end
    if not can_farm() then
        farm_target = nil
        was_down = false
        set_farm_noclip(false)
        return
    end
    local hrp = farm_hrp()
    if not hrp then return end

    if coin_bags_full() then
        farm_target = nil
        farm_release()
        return
    end

    local list = available_coins()
    if #list > 0 then
        if not coin_available(farm_target) then
            farm_target = nearest_coin(hrp.Position, list)
        end

        if farm_target then
            set_farm_noclip(true)
            local cpos = farm_target.Position
            down_ref_y = cpos.Y
            local dest = cpos

            if farm_mode == "Down" then
                local xz = flat_dist(hrp.Position, cpos)
                if xz <= DOWN_RISE_XZ then
                    dest = cpos
                    fire_touch(farm_target)
                else
                    dest = Vector3.new(cpos.X, cpos.Y - DOWN_DEPTH, cpos.Z)
                end
            elseif (cpos - hrp.Position).Magnitude <= 6 then
                fire_touch(farm_target)
            end

            farm_move(hrp, dest, dt)
        end
    else
        farm_target = nil
        farm_release()
    end
end)

local SpooferMainGroup = Tabs.Spoofer:AddLeftGroupbox('Profile Settings')

local CFG = {
    username         = "xX_Shadow_Xx",
    level            = "999",
    prestige         = "X",
    xpBarFill        = 1.0,
    murdererElims    = "9999",
    sheriffVictories = "999",
    innocentSurvivals= "999",
    enabled          = true,
}

local function applyText(inst, text)
    if inst and (inst:IsA("TextLabel") or inst:IsA("TextButton")) then
        if inst.Text ~= text then inst.Text = text end
    end
end

local activeConns = {}

local function clearConns()
    for _, c in ipairs(activeConns) do pcall(function() c:Disconnect() end) end
    table.clear(activeConns)
end

local function watch(inst, desired)
    if not inst then return end
    local conn = inst:GetPropertyChangedSignal("Text"):Connect(function()
        if CFG.enabled and inst.Text ~= desired then inst.Text = desired end
    end)
    table.insert(activeConns, conn)
end

local function spoofLeaderboardFrame(frame)
    if not CFG.enabled then return end
    local lbl = frame:FindFirstChild("PlayerLabel")
    if lbl then applyText(lbl, CFG.username); watch(lbl, CFG.username) end

    local lvlImg = frame:FindFirstChild("Level")
    if lvlImg then
        local lvl  = lvlImg:FindFirstChild("Level")
        local pres = lvlImg:FindFirstChild("Prestige")
        if lvl  then applyText(lvl,  CFG.level);    watch(lvl,  CFG.level)   end
        if pres then applyText(pres, CFG.prestige); watch(pres, CFG.prestige) end
    end

    local bar = frame:FindFirstChild("XPBar") and frame.XPBar:FindFirstChild("Bar")
    if bar then bar.Size = UDim2.new(CFG.xpBarFill, 0, 1, 0) end
end

local function spoofViewProfile(mainGui)
    if not CFG.enabled then return end
    local ok, vp = pcall(function()
        return mainGui.Game.ViewProfile.Main.Profile.Container
    end)
    if not ok or not vp then return end

    -- Username header
    local uname = vp:FindFirstChild("Username")
    if uname then applyText(uname, CFG.username .. " (You)"); watch(uname, CFG.username .. " (You)") end

    -- CharacterIcon level
    local ci = vp.Character and vp.Character:FindFirstChild("CharacterIcon")
    if ci then
        local lc = ci:FindFirstChild("LevelFrame") and ci.LevelFrame:FindFirstChild("LevelContainer")
        if lc then
            local ico = lc:FindFirstChild("Icon")
            if ico then
                local lvlTL = ico:FindFirstChild("Level")
                if lvlTL then applyText(lvlTL, CFG.level); watch(lvlTL, CFG.level) end
            end
            local presTL = lc:FindFirstChild("Prestige")
            if presTL then applyText(presTL, CFG.prestige); watch(presTL, CFG.prestige) end
        end
    end

    -- LevelOld
    local lo = vp.Character and vp.Character:FindFirstChild("LevelOld")
    if lo then
        local loL = lo:FindFirstChild("Level")
        if loL then
            local loLI = loL:FindFirstChild("Level")
            if loLI then
                local loTL = loLI:FindFirstChild("Level")
                if loTL then applyText(loTL, CFG.level); watch(loTL, CFG.level) end
            end
            local loP = loL:FindFirstChild("Prestige")
            if loP then applyText(loP, CFG.prestige); watch(loP, CFG.prestige) end
        end
    end

    -- Season1Stats
    local s1 = vp.Profile and vp.Profile:FindFirstChild("Season1Stats")
    if s1 then
        local ct = s1:FindFirstChild("Container")
        if ct then
            local function patchStat(childName, val)
                local node = ct:FindFirstChild(childName)
                if node then
                    local amt = node:FindFirstChild("Amount")
                    if amt then applyText(amt, val); watch(amt, val) end
                end
            end
            patchStat("Eliminations", CFG.murdererElims)
            patchStat("Saves",        CFG.sheriffVictories)
            patchStat("Survivals",    CFG.innocentSurvivals)
        end
    end

    -- Tab titles
    local main = vp.Parent and vp.Parent.Parent
    if main then
        for _, tabName in ipairs({"Effects","Emotes","Perks","Pets","Radios","Weapons"}) do
            local tab = main:FindFirstChild(tabName)
            local ul = tab and tab:FindFirstChild("Title") and tab.Title:FindFirstChild("Username")
            if ul then
                local desired = CFG.username .. "'s " .. tabName
                applyText(ul, desired); watch(ul, desired)
            end
        end
    end
end

local function hookMainGui(mainGui)
    clearConns()

    -- Leaderboard list
    local ok, container = pcall(function() return mainGui.Game.Leaderboard.Container end)
    if ok and container then
        for _, child in ipairs(container:GetChildren()) do
            if child:IsA("Frame") and child:FindFirstChild("IsPlayerFrame") and child.Name == lp.Name then
                spoofLeaderboardFrame(child)
            end
        end
        local ca = container.ChildAdded:Connect(function(child)
            task.wait(0.15)
            if child:IsA("Frame") and child:FindFirstChild("IsPlayerFrame") and child.Name == lp.Name then
                spoofLeaderboardFrame(child)
            end
        end)
        table.insert(activeConns, ca)
    end

    -- ViewProfile: hook Visible changes
    local okVP, vpFrame = pcall(function() return mainGui.Game.ViewProfile end)
    if okVP and vpFrame then
        spoofViewProfile(mainGui)
        local vc = vpFrame:GetPropertyChangedSignal("Visible"):Connect(function()
            if vpFrame.Visible then
                task.wait(0.05)
                spoofViewProfile(mainGui)
            end
        end)
        table.insert(activeConns, vc)
    end
end

local function reapplySpoof()
    clearConns()
    local pg = lp:FindFirstChild("PlayerGui")
    local mg = pg and pg:FindFirstChild("MainGUI")
    if mg then hookMainGui(mg) end
end

local pg = lp:WaitForChild("PlayerGui")
local existingMain = pg:FindFirstChild("MainGUI")
if existingMain then hookMainGui(existingMain) end

local pgAddedConn = pg.ChildAdded:Connect(function(child)
    if child.Name == "MainGUI" then
        task.wait(0.3)
        hookMainGui(child)
    end
end)
