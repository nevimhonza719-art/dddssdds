-- CORE HELPERS & SMOOTH TWEEN ENGINE

local function TrackConnection(conn)
    table.insert(Library.Connections, conn)
    return conn
end

local function Tween(inst, duration, props, easingStyle, easingDir)
    if not inst then return end
    local t = TweenService:Create(
        inst,
        TweenInfo.new(
            duration or 0.18,
            easingStyle or Enum.EasingStyle.Quart,
            easingDir or Enum.EasingDirection.Out
        ),
        props
    )
    t:Play()
    return t
end

local function Create(className, props, children)
    local inst = Instance.new(className)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then
            inst[k] = v
        end
    end
    for _, child in ipairs(children or {}) do
        child.Parent = inst
    end
    if props and props.Parent then
        inst.Parent = props.Parent
    end
    return inst
end

local function AddCorner(parent, radius)
    return Create("UICorner", {
        CornerRadius = UDim.new(0, radius or 4),
        Parent = parent
    })
end

local function AddStroke(parent, color, thickness, transparency)
    return Create("UIStroke", {
        Color = color or Library.Theme.Border,
        Thickness = thickness or 1,
        Transparency = transparency or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
        Parent = parent
    })
end

local function AddPadding(parent, top, bottom, left, right)
    return Create("UIPadding", {
        PaddingTop = UDim.new(0, top or 0),
        PaddingBottom = UDim.new(0, bottom or top or 0),
        PaddingLeft = UDim.new(0, left or top or 0),
        PaddingRight = UDim.new(0, right or left or top or 0),
        Parent = parent
    })
end

local function GetSafeGuiParent()
    local ok, res = pcall(function()
        if gethui then
            return gethui()
        elseif CoreGui:FindFirstChild("RobloxGui") then
            return CoreGui
        end
    end)
    if ok and res then
        return res
    end
    if LocalPlayer then
        return LocalPlayer:WaitForChild("PlayerGui")
    end
    return CoreGui
end

        -- 3. ADD SLIDER
        function Groupbox:AddSlider(idx, info)
            if type(idx) == "table" then
                info = idx
                idx = info.Flag or info.Text or ("Slider_" .. tostring(#Groupbox.Items + 1))
            end
            info = info or {}
            local text = info.Text or idx
            local min = info.Min or 0
            local max = info.Max or 100
            local rounding = info.Rounding or 0
            local suffix = info.Suffix or ""
            local val = math.clamp(info.Default or min, min, max)

            local SliderObj = {
                Value = val,
                Type = "Slider",
                ChangedCallbacks = {}
            }

            local Wrap = Create("Frame", {
                Name = "Slider_" .. text,
                Size = UDim2.new(1, 0, 0, 38),
                BackgroundTransparency = 1,
                Parent = ItemsContainer
            })

            Create("TextLabel", {
                Size = UDim2.new(0.65, 0, 0, 16),

                Font = Enum.Font.GothamMedium,
                Text = text,
                TextColor3 = Library.Theme.TextInactive,
                TextSize = 13,
                TextXAlignment = Enum.TextXAlignment.Left,
                Parent = Wrap
            })

            local ValueLabel = Create("TextLabel", {
                AnchorPoint = Vector2.new(1, 0),
                Position = UDim2.new(1, 0, 0, 0),
                Size = UDim2.new(0.35, 0, 0, 16),
                BackgroundTransparency = 1,
                Font = Enum.Font.GothamBold,
                Text = tostring(val) .. suffix,
                TextColor3 = Library.Theme.TextActive,
                TextSize = 12,
                TextXAlignment = Enum.TextXAlignment.Right,
                Parent = Wrap
            })

            local Track = Create("TextButton", {
                Position = UDim2.fromOffset(0, 24),
                Size = UDim2.new(1, 0, 0, 6),
                BackgroundColor3 = Library.Theme.SwitchTrackOff,
                BorderSizePixel = 0,
                Text = "",
                AutoButtonColor = false,
                Parent = Wrap
            })
            AddCorner(Track, 99)

            local Fill = Create("Frame", {
                Size = UDim2.new((val - min) / math.max(max - min, 0.0001), 0, 1, 0),
                BackgroundColor3 = Library.Theme.Accent,
                BorderSizePixel = 0,
                Parent = Track
            })
            AddCorner(Fill, 99)

            local Knob = Create("Frame", {
                AnchorPoint = Vector2.new(0.5, 0.5),
                Position = UDim2.new(1, 0, 0.5, 0),
                Size = UDim2.fromOffset(11, 11),
                BackgroundColor3 = Color3.fromRGB(255, 255, 255),
                BorderSizePixel = 0,
                Parent = Fill
            })
            AddCorner(Knob, 99)

            local function Round(n, dec)
                local mult = 10 ^ (dec or 0)
                return math.floor(n * mult + 0.5) / mult
            end

            local function SetValue(newVal, skipCallback)
                newVal = math.clamp(Round(newVal, rounding), min, max)
                val = newVal
                SliderObj.Value = val
                local alpha = (val - min) / math.max(max - min, 0.0001)
                Tween(Fill, 0.08, { Size = UDim2.new(alpha, 0, 1, 0) })
                ValueLabel.Text = tostring(val) .. suffix
                if not skipCallback then
                    if info.Callback then task.spawn(info.Callback, val) end
                    for _, cb in ipairs(SliderObj.ChangedCallbacks) do
                        task.spawn(cb, val)
                    end
                end
            end

            local draggingSlider = false
            local function UpdateSlider(input)
                local alpha = math.clamp((input.Position.X - Track.AbsolutePosition.X) / Track.AbsoluteSize.X, 0, 1)
                SetValue(min + (max - min) * alpha)
            end

            Track.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    draggingSlider = true
                    UpdateSlider(input)
                end
            end)
            TrackConnection(UserInputService.InputEnded:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    draggingSlider = false
                end
            end))
            TrackConnection(UserInputService.InputChanged:Connect(function(input)
                if draggingSlider and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                    UpdateSlider(input)
                end
            end))

            function SliderObj:SetValue(v) SetValue(v) end
            function SliderObj:OnChanged(cb) table.insert(SliderObj.ChangedCallbacks, cb) end

            Options[idx] = SliderObj
            table.insert(Groupbox.Items, { Frame = Wrap, SearchText = text })
            return SliderObj
        end

                Size = UDim2.fromOffset(200, 0),

            local function GetSelectedString()
                if not multi then return tostring(selected or "None") end
                local list = {}
                for _, v in ipairs(values) do
                    if selected[v] then table.insert(list, tostring(v)) end
                end
                return #list > 0 and table.concat(list, ", ") or "None"
            end

            local function Rebuild()
                for _, c in ipairs(MenuFrame:GetChildren()) do
                    if c:IsA("TextButton") then c:Destroy() end
                end
                SelectedLabel.Text = GetSelectedString()

                for _, optName in ipairs(values) do
                    local isChosen = multi and (selected[optName] == true) or (selected == optName)
                    local optBtn = Create("TextButton", {
                        Size = UDim2.new(1, 0, 0, 26),
                        BackgroundColor3 = isChosen and Library.Theme.TabActiveBg or Color3.fromRGB(14, 14, 18),

                        TextXAlignment = Enum.TextXAlignment.Left,
                        AutoButtonColor = false,
                        ZIndex = 106,
                        Parent = MenuFrame
                    })
                    AddCorner(optBtn, 4)

                    optBtn.MouseButton1Click:Connect(function()
                        if multi then
                            selected[optName] = not selected[optName]
                        else
                            selected = optName
                            CloseActivePopover()
                        end
                        DropdownObj.Value = selected
                        Rebuild()
                        if info.Callback then task.spawn(info.Callback, selected) end
                        for _, cb in ipairs(DropdownObj.ChangedCallbacks) do
                            task.spawn(cb, selected)
                        end
                    end)
                end
            end

            DropBtn.MouseButton1Click:Connect(function()
                if ActivePopover == MenuFrame and MenuFrame.Visible then
                    CloseActivePopover()
                    return
                end
                CloseActivePopover()
                Rebuild()
                local pos = DropBtn.AbsolutePosition
                MenuFrame.Position = UDim2.fromOffset(pos.X, pos.Y + 35)
                MenuFrame.Size = UDim2.fromOffset(DropBtn.AbsoluteSize.X, math.min(#values * 28 + 8, 150))
                MenuFrame.Visible = true
                ActivePopover = MenuFrame
            end)

            Rebuild()

            function DropdownObj:SetValue(val)
                selected = val
                DropdownObj.Value = selected
                Rebuild()
            end
            function DropdownObj:SetValues(newValues)
                values = newValues or {}
                DropdownObj.Values = values
                Rebuild()
            end
            function DropdownObj:OnChanged(cb)
                table.insert(DropdownObj.ChangedCallbacks, cb)
            end

            Options[idx] = DropdownObj
            table.insert(Groupbox.Items, { Frame = Wrap, SearchText = text })
            return DropdownObj
        end

            Parent = TabViewsHolder
        })

        -- Horizontal Sub-Tab Bar (Matches Cosmetics | Gameplay | Emotes | UI | Statistics)
        local SubTabBar = Create("Frame", {
            Name = "SubTabBar",
            Position = UDim2.fromOffset(22, 12),
            Size = UDim2.new(1, -44, 0, 36),
            BackgroundTransparency = 1,
            Visible = false,
            Parent = TabView
        })

        local SubTabButtonsRow = Create("Frame", {
            Name = "ButtonsRow",
            Size = UDim2.fromOffset(0, 34),
            AutomaticSize = Enum.AutomaticSize.X,
            BackgroundTransparency = 1,
            Parent = SubTabBar
        }, {
            Create("UIListLayout", {
                FillDirection = Enum.FillDirection.Horizontal,
                VerticalAlignment = Enum.VerticalAlignment.Center,
                SortOrder = Enum.SortOrder.LayoutOrder,
                Padding = UDim.new(0, 6)
            })
        })

        -- 2px Baseline directly underneath the Sub-Tabs row (matches screenshot width & color)
        local SubTabBaseline = Create("Frame", {
            Name = "Baseline",
            Position = UDim2.fromOffset(0, 34),
            Size = UDim2.fromOffset(480, 2),
            BackgroundColor3 = Library.Theme.SubTabBaseline,
            BorderSizePixel = 0,
            Parent = SubTabBar
        })

        -- Active sub-tab dark break / indicator on the baseline (exact match to "Gameplay" break in screenshot)
        local SubTabActiveSegment = Create("Frame", {
            Name = "ActiveSegment",
            Position = UDim2.fromOffset(0, 0),
            Size = UDim2.fromOffset(96, 2),
            BackgroundColor3 = Library.Theme.SubTabActiveLine,
            BorderSizePixel = 0,
            Parent = SubTabBaseline
        })

        -- Keep baseline width synced to the width of all sub-tabs
        SubTabButtonsRow:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
            SubTabBaseline.Size = UDim2.fromOffset(SubTabButtonsRow.AbsoluteSize.X, 2)
        end)

        local SubTabPagesHolder = Create("Frame", {
            Name = "SubTabPages",
            Position = UDim2.fromOffset(0, 0),
            Size = UDim2.fromScale(1, 1),
            BackgroundTransparency = 1,
            Parent = TabView
        })

        function Tab:Select()
            CloseActivePopover()
            for _, t in ipairs(Window.Tabs) do
                t.View.Visible = false
                Tween(t.Button, 0.16, { BackgroundTransparency = 1 })
                Tween(t.Label, 0.16, { TextColor3 = Library.Theme.TextInactive })
                t.Icon:SetColor(Library.Theme.TextInactive, 0.16)
                t.Label.Font = Enum.Font.GothamMedium
            end

            Window.ActiveTab = Tab
            TabView.Visible = true
            TopHeaderTitle.Text = tabName

            Tween(TabBtn, 0.16, { BackgroundTransparency = 0 })
            Tween(TabLabel, 0.16, { TextColor3 = Library.Theme.TextActive })
            TabIcon:SetColor(Library.Theme.Accent, 0.16)
            TabLabel.Font = Enum.Font.GothamBold
        end

        TabBtn.MouseButton1Click:Connect(function()
            Tab:Select()
        end)

        -- Add Horizontal Sub-Tab
        function Tab:AddSubTab(subConfig)
            subConfig = type(subConfig) == "string" and { Name = subConfig } or (subConfig or {})
            local subName = subConfig.Name or subConfig.Title or "SubTab"
            local subIconName = subConfig.Icon or string.lower(subName)

            SubTabBar.Visible = true
            SubTabPagesHolder.Position = UDim2.fromOffset(0, 50)
            SubTabPagesHolder.Size = UDim2.new(1, 0, 1, -50)

            local SubTab = {
                Name = subName
            }

            local SubBtn = Create("TextButton", {
                Name = "SubTab_" .. subName,
                Size = UDim2.fromOffset(0, 32),
                AutomaticSize = Enum.AutomaticSize.X,
                BackgroundColor3 = Library.Theme.TabActiveBg,
                BackgroundTransparency = 1,
                BorderSizePixel = 0,
                Text = "",
                AutoButtonColor = false,
                Parent = SubTabButtonsRow
            }, {
                Create("UIListLayout", {
                    FillDirection = Enum.FillDirection.Horizontal,
                    VerticalAlignment = Enum.VerticalAlignment.Center,
                    SortOrder = Enum.SortOrder.LayoutOrder,
                    Padding = UDim.new(0, 7)
                }),
                AddPadding(nil, 0, 0, 12, 12)
            })
            AddCorner(SubBtn, 4)

            local SubIcon = BuildVectorIcon(subIconName, 14, Library.Theme.TextInactive, SubBtn)
            local SubLabel = Create("TextLabel", {
                Size = UDim2.fromOffset(0, 32),
                AutomaticSize = Enum.AutomaticSize.X,
                BackgroundTransparency = 1,
                Font = Enum.Font.GothamMedium,
                Text = subName,
                TextColor3 = Library.Theme.TextInactive,
                TextSize = 13,
                Parent = SubBtn
            })

            SubBtn.MouseEnter:Connect(function()
                if Tab.ActiveSubTab ~= SubTab then
                    Tween(SubLabel, 0.14, { TextColor3 = Color3.fromRGB(185, 185, 195) })
                    SubIcon:SetColor(Color3.fromRGB(165, 165, 175), 0.14)
                end
            end)
            SubBtn.MouseLeave:Connect(function()
                if Tab.ActiveSubTab ~= SubTab then
                    Tween(SubLabel, 0.14, { TextColor3 = Library.Theme.TextInactive })
                    SubIcon:SetColor(Library.Theme.TextInactive, 0.14)
                end
            end)

            local ScrollPage, LeftCol, RightCol = CreateColumnsPage(SubTabPagesHolder)
            SubTab.Page = ScrollPage
            SubTab.LeftCol = LeftCol
            SubTab.RightCol = RightCol

            function SubTab:Select()
                CloseActivePopover()
                for _, st in ipairs(Tab.SubTabs) do
                    st.Page.Visible = false
                    Tween(st.Button, 0.16, { BackgroundTransparency = 1 })
                    Tween(st.Label, 0.16, { TextColor3 = Library.Theme.TextInactive })
                    st.Icon:SetColor(Library.Theme.TextInactive, 0.16)
                    st.Label.Font = Enum.Font.GothamMedium
                end

                Tab.ActiveSubTab = SubTab
                ScrollPage.Visible = true
                Tween(SubBtn, 0.16, { BackgroundTransparency = 0 })
                Tween(SubLabel, 0.16, { TextColor3 = Library.Theme.TextActive })
                SubIcon:SetColor(Library.Theme.Accent, 0.16)
                SubLabel.Font = Enum.Font.GothamBold

                task.defer(function()
                    SubTabBaseline.Size = UDim2.fromOffset(SubTabButtonsRow.AbsoluteSize.X, 2)
                    local relX = SubBtn.AbsolutePosition.X - SubTabBaseline.AbsolutePosition.X
                    local w = math.max(SubBtn.AbsoluteSize.X, 40)
                    Tween(SubTabActiveSegment, 0.2, {
                        Position = UDim2.fromOffset(math.max(0, relX), 0),
                        Size = UDim2.fromOffset(w, 2)
                    })
                end)
            end

            SubBtn.MouseButton1Click:Connect(function()
                SubTab:Select()
            end)

            function SubTab:AddLeftGroupbox(gbConfig)
                return CreateGroupboxCard(LeftCol, gbConfig)
            end
            function SubTab:AddRightGroupbox(gbConfig)
                return CreateGroupboxCard(RightCol, gbConfig)
            end

            SubTab.Button = SubBtn
            SubTab.Label = SubLabel
            SubTab.Icon = SubIcon
            table.insert(Tab.SubTabs, SubTab)

            if #Tab.SubTabs == 1 then
                SubTab:Select()
            end

            return SubTab
        end

        local function EnsureDefaultPage()
            if not Tab.DefaultPage then
                local ScrollPage, LeftCol, RightCol = CreateColumnsPage(SubTabPagesHolder)
                ScrollPage.Visible = true
                Tab.DefaultPage = { Page = ScrollPage, LeftCol = LeftCol, RightCol = RightCol }
            end
            return Tab.DefaultPage
        end

        function Tab:AddLeftGroupbox(gbConfig)
            return CreateGroupboxCard(EnsureDefaultPage().LeftCol, gbConfig)
        end
        function Tab:AddRightGroupbox(gbConfig)
            return CreateGroupboxCard(EnsureDefaultPage().RightCol, gbConfig)
        end

        Tab.View = TabView
        Tab.Button = TabBtn
        Tab.Label = TabLabel
        Tab.Icon = TabIcon
        table.insert(Window.Tabs, Tab)

        if #Window.Tabs == 1 then
            Tab:Select()
        end

        return Tab
    end

    -- Keyboard Shortcuts (Ctrl+K for search, RightControl to toggle UI)
    TrackConnection(UserInputService.InputBegan:Connect(function(input, gpe)
        if input.KeyCode == Enum.KeyCode.K and UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
            if MainFrame.Visible then
                SearchInput:CaptureFocus()
            end
        elseif not gpe and input.KeyCode == Library.ToggleKeybind then
            Window.Visible = not Window.Visible
            MainFrame.Visible = Window.Visible
            CloseActivePopover()
        end
    end))

    table.insert(Library.Windows, Window)
    return Window
end

function Library:Unload()
    for _, conn in ipairs(Library.Connections) do
        pcall(function() conn:Disconnect() end)
    end
    if ScreenGui then
        ScreenGui:Destroy()
    end
end

return Library
