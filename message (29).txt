-- Utility functions
local function Tween(instance, duration, props, style, dir)
    local t = TweenService:Create(
        instance,
        TweenInfo.new(duration or 0.16, style or Enum.EasingStyle.Quart, dir or Enum.EasingDirection.Out),
        props
    )
    t:Play()
    return t
end

local function Create(className, props, children)
    local inst = Instance.new(className)
    for k, v in pairs(props or {}) do
        inst[k] = v
    end
    for _, child in ipairs(children or {}) do
        child.Parent = inst
    end
    return inst
end

local function AddCorner(parent, radius)
    return Create("UICorner", {
        CornerRadius = UDim.new(0, radius or 6),
        Parent = parent
    })
end

local function AddStroke(parent, color, thickness, transparency)
    return Create("UIStroke", {
        Color = color or Library.Theme.Border,
        Thickness = thickness or 1,
        Transparency = transparency or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
        Parent = parent
    })
end

local function AddPadding(parent, top, bottom, left, right)
    return Create("UIPadding", {
        PaddingTop = UDim.new(0, top or 0),
        PaddingBottom = UDim.new(0, bottom or top or 0),
        PaddingLeft = UDim.new(0, left or top or 0),
        PaddingRight = UDim.new(0, right or left or top or 0),
        Parent = parent
    })
end

local function ResolveIcon(iconName)
    if not iconName or iconName == "" then
        return nil
    end
    if string.find(iconName, "rbxassetid://") or string.find(iconName, "http") then
        return iconName
    end
    return Library.Icons[string.lower(iconName)] or Library.Icons["layers"]
end

local function GetSafeGuiParent()
    local success, parent = pcall(function()
        if gethui then
            return gethui()
        elseif CoreGui:FindFirstChild("RobloxGui") then
            return CoreGui
        end
    end)
    if success and parent then
        return parent
    end
    if LocalPlayer then
        return LocalPlayer:WaitForChild("PlayerGui")
    end
    return CoreGui
end

-- Create an Icon Widget (ImageLabel with clean fallback)
local function CreateIcon(iconName, size, color, parent)
    local asset = ResolveIcon(iconName)
    local holder = Create("Frame", {
        BackgroundTransparency = 1,
        Size = UDim2.fromOffset(size or 16, size or 16),
        Parent = parent
    })

    local img = Create("ImageLabel", {
        Name = "IconImage",
        BackgroundTransparency = 1,
        Size = UDim2.fromScale(1, 1),
        Image = asset or "",
        ImageColor3 = color or Library.Theme.SubText,
        ScaleType = Enum.ScaleType.Fit,
        Parent = holder
    })

    return holder, img
end

    AddStroke(card, Library.Theme.Border, 1)

    Create("Frame", {
        Size = UDim2.new(0, 3, 1, 0),
        BackgroundColor3 = Library.Theme.Accent,
        BorderSizePixel = 0,
        Parent = card
    })

    Create("TextLabel", {
        Position = UDim2.fromOffset(14, 10),
        Size = UDim2.new(1, -24, 0, 16),
        BackgroundTransparency = 1,
        Font = Enum.Font.GothamBold,
        Text = title,
        TextColor3 = Library.Theme.Text,
        TextSize = 13,
        TextXAlignment = Enum.TextXAlignment.Left,
        Parent = card
    })

    Create("TextLabel", {
        Position = UDim2.fromOffset(14, 28),

        Font = Enum.Font.Gotham,

        if card and card.Parent then
            Tween(card, 0.2, { BackgroundTransparency = 1 })
            task.wait(0.22)
            if card then card:Destroy() end
        end
    end)

    -- Filter groupboxes/elements when typing in Search
    local function ApplySearchFilter(query)
        query = string.lower(query or "")
        for _, gb in ipairs(Window.AllGroupboxes) do
            if query == "" then
                gb.Frame.Visible = true
                for _, item in ipairs(gb.Items) do
                    item.Frame.Visible = true
                end
            else
                local gbMatches = string.find(string.lower(gb.Name), query, 1, true) ~= nil
                local anyItemMatches = false
                for _, item in ipairs(gb.Items) do
                    local match = gbMatches or (string.find(string.lower(item.SearchText or ""), query, 1, true) ~= nil)
                    item.Frame.Visible = match
                    if match then
                        anyItemMatches = true
                    end
                end
                gb.Frame.Visible = gbMatches or anyItemMatches
            end
        end
    end

                values = newValues or {}
                DropdownObj.Values = values
                RebuildOptions()
            end
            function DropdownObj:OnChanged(cb)
                table.insert(DropdownObj.ChangedCallbacks, cb)
            end

            Options[idx] = DropdownObj
            table.insert(Groupbox.Items, { Frame = Wrap, SearchText = text })
            return DropdownObj
        end

        function Tab:AddSubTab(subConfig)
            subConfig = type(subConfig) == "string" and { Name = subConfig } or (subConfig or {})
            local subName = subConfig.Name or subConfig.Title or "SubTab"
            local subIcon = subConfig.Icon or string.lower(subName)

            SubTabBar.Visible = true
            SubTabPagesHolder.Position = UDim2.fromOffset(0, 42)
            SubTabPagesHolder.Size = UDim2.new(1, 0, 1, -42)

            local SubTab = {
                Name = subName
            }

            local SubBtn = Create("TextButton", {
                Name = "SubTab_" .. subName,
                Size = UDim2.fromOffset(0, 36),
                AutomaticSize = Enum.AutomaticSize.X,
                BackgroundTransparency = 1,
                Text = "",
                AutoButtonColor = false,
                Parent = SubTabButtonsHolder
            }, {
                Create("UIListLayout", {
                    FillDirection = Enum.FillDirection.Horizontal,
                    VerticalAlignment = Enum.VerticalAlignment.Center,
                    SortOrder = Enum.SortOrder.LayoutOrder,
                    Padding = UDim.new(0, 7)
                }),
                AddPadding(nil, 0, 0, 4, 4)
            })

            local _, SubIconImg = CreateIcon(subIcon, 15, Library.Theme.SubText, SubBtn)
            local SubLabel = Create("TextLabel", {
                Size = UDim2.fromOffset(0, 36),
                AutomaticSize = Enum.AutomaticSize.X,
                BackgroundTransparency = 1,
                Font = Enum.Font.GothamMedium,
                Text = subName,
                TextColor3 = Library.Theme.SubText,
                TextSize = 13,
                Parent = SubBtn
            })

            local ScrollPage, LeftCol, RightCol = CreateColumnsPage(SubTabPagesHolder)
            SubTab.Page = ScrollPage
            SubTab.LeftCol = LeftCol
            SubTab.RightCol = RightCol

            function SubTab:Select()
                CloseActivePopover()
                for _, st in ipairs(Tab.SubTabs) do
                    st.Page.Visible = false
                    Tween(st.Label, 0.15, { TextColor3 = Library.Theme.SubText })
                    Tween(st.IconImg, 0.15, { ImageColor3 = Library.Theme.SubText })

                Tab.ActiveSubTab = SubTab
                ScrollPage.Visible = true
                Tween(SubLabel, 0.15, { TextColor3 = Library.Theme.Text })
                Tween(SubIconImg, 0.15, { ImageColor3 = Library.Theme.Accent })
                SubLabel.Font = Enum.Font.GothamBold

                task.defer(function()
                    local relX = SubBtn.AbsolutePosition.X - SubTabBaseline.AbsolutePosition.X
                    local width = math.max(SubBtn.AbsoluteSize.X, 50)
                    Tween(SubTabUnderline, 0.2, {
                        Position = UDim2.fromOffset(math.max(0, relX), 0),
                        Size = UDim2.fromOffset(width, 2)
                    })
                end)
            end

            SubBtn.MouseButton1Click:Connect(function()
                SubTab:Select()
            end)

                return CreateGroupboxInColumn(LeftCol, gbConfig)
            end

            function SubTab:AddRightGroupbox(gbConfig)
                return CreateGroupboxInColumn(RightCol, gbConfig)
            end

            SubTab.Button = SubBtn
            SubTab.Label = SubLabel
            SubTab.IconImg = SubIconImg
            table.insert(Tab.SubTabs, SubTab)

            if #Tab.SubTabs == 1 then
                SubTab:Select()
            end

        end

        -- Direct Groupbox methods on Tab (if the user doesn't need Sub-Tabs on a tab)
        local function EnsureDefaultPage()
            if not Tab.DefaultPage then
                local ScrollPage, LeftCol, RightCol = CreateColumnsPage(SubTabPagesHolder)
                ScrollPage.Visible = true
                Tab.DefaultPage = { Page = ScrollPage, LeftCol = LeftCol, RightCol = RightCol }
            end
            return Tab.DefaultPage
        end

        function Tab:AddLeftGroupbox(gbConfig)
            return CreateGroupboxInColumn(EnsureDefaultPage().LeftCol, gbConfig)
        end

        function Tab:AddRightGroupbox(gbConfig)
            return CreateGroupboxInColumn(EnsureDefaultPage().RightCol, gbConfig)
        end

        Tab.View = TabView
        Tab.Button = TabBtn
        Tab.Label = TabLabel
        Tab.IconImg = IconImg
        table.insert(Window.Tabs, Tab)

        if #Window.Tabs == 1 then
            Tab:Select()
        end

        return Tab
    end

    -- Keyboard Shortcuts: RightControl (or custom) to Toggle Window, Ctrl+K to focus Search
    UserInputService.InputBegan:Connect(function(input, gpe)
        if input.KeyCode == Enum.KeyCode.K and UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
            if MainFrame.Visible then
                SearchInput:CaptureFocus()
            end
        elseif not gpe and input.KeyCode == Library.ToggleKeybind then
            Window.Visible = not Window.Visible
            MainFrame.Visible = Window.Visible
            CloseActivePopover()
        end
    end)

    table.insert(Library.Windows, Window)
    return Window
end

-- Save / Load Configuration Helper
function Library:GetConfig()
    local data = { Toggles = {}, Options = {} }
    for k, v in pairs(Toggles) do
        data.Toggles[k] = v.Value
    end
    for k, v in pairs(Options) do
        if type(v.Value) == "Color3" then
            data.Options[k] = { R = v.Value.R, G = v.Value.G, B = v.Value.B, __type = "Color3" }
        elseif typeof(v.Value) == "EnumItem" then
            data.Options[k] = { Name = v.Value.Name, __type = "KeyCode" }
        else
            data.Options[k] = v.Value
        end
    end
    return HttpService:JSONEncode(data)
end

function Library:LoadConfig(jsonString)
    local ok, data = pcall(function()
        return HttpService:JSONDecode(jsonString)
    end)
    if not ok or type(data) ~= "table" then return false end

    for k, val in pairs(data.Toggles or {}) do
        if Toggles[k] and Toggles[k].SetValue then
            Toggles[k]:SetValue(val)
        end
    end
    for k, val in pairs(data.Options or {}) do
        if Options[k] then
            if type(val) == "table" and val.__type == "Color3" and Options[k].SetValueRGB then
                Options[k]:SetValueRGB(Color3.new(val.R, val.G, val.B))
            elseif Options[k].SetValue then
                Options[k]:SetValue(val)
            end
        end
    end
    return true
end

function Library:Unload()
    Library.Unloaded = true
    if ScreenGui then
        ScreenGui:Destroy()
    end
end

return Library
