local function EnsureFolders()

        if makefolder then

            end
            if not isfolder(Library.FolderName .. "/configs") then
                makefolder(Library.FolderName .. "/configs")
            end
            if not isfolder(Library.FolderName .. "/assets") then
                makefolder(Library.FolderName .. "/assets")
            end
        end
    end)
end

-- Cache remote PNG (such as https://files.catbox.moe/0cd26t.png) via getcustomasset if executor supports it
local CustomAssetCache = {}
local function GetExternalAsset(url, fileName)
    if CustomAssetCache[url] ~= nil then
        return CustomAssetCache[url]
    end
    local getAsset = (getcustomasset or getsynasset)
    if getAsset and writefile and isfile then
        local ok, assetId = pcall(function()
            local path = Library.FolderName .. "/assets/" .. fileName
            if not isfile(path) then
                local content = game:HttpGet(url)
                if content and #content > 0 then
                    writefile(path, content)
                end
            end
            if isfile(path) then
                return getAsset(path)
            end
        end)
        if ok and assetId then
            CustomAssetCache[url] = assetId
            return assetId
        end
    end
    CustomAssetCache[url] = false
    return nil
end

local function TrackConn(conn)
    table.insert(Library.Connections, conn)
    return conn
end

local function Tween(inst, duration, props, style, dir)
    if not inst then return end
    local t = TweenService:Create(
        inst,
        TweenInfo.new(
            duration or Library.TweenSpeed,
            style or Library.EasingStyle,
            dir or Enum.EasingDirection.Out
        ),
        props
    )
    t:Play()
    return t
end

local function FastTween(inst, duration, props)
    return Tween(inst, duration or 0.16, props, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
end

local function Create(className, props, children)
    local inst = Instance.new(className)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then
            inst[k] = v
        end
    end
    for _, child in ipairs(children or {}) do
        child.Parent = inst
    end
    if props and props.Parent then
        inst.Parent = props.Parent
    end
    return inst
end

-- Smooth Lerped Dragging Binder
local function MakeDraggable(handle, targetFrame, onDragStart)
    local isDragging = false
    local dragStartMouse = Vector2.zero
    local dragStartPos = targetFrame.Position
    local desiredPos = targetFrame.Position

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            isDragging = true
            dragStartMouse = Vector2.new(input.Position.X, input.Position.Y)
            dragStartPos = targetFrame.Position
            desiredPos = targetFrame.Position
            if onDragStart then onDragStart() end

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    isDragging = false
                end
            end)
        end
    end)

    TrackConn(UserInputService.InputChanged:Connect(function(input)
        if isDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = Vector2.new(input.Position.X, input.Position.Y) - dragStartMouse
            desiredPos = UDim2.new(
                dragStartPos.X.Scale,
                dragStartPos.X.Offset + delta.X,
                dragStartPos.Y.Scale,
                dragStartPos.Y.Offset + delta.Y
            )
        end
    end))

    TrackConn(RunService.RenderStepped:Connect(function(dt)
        if targetFrame and targetFrame.Parent then
            targetFrame.Position = targetFrame.Position:Lerp(desiredPos, math.clamp(dt * 22, 0, 1))
        end
    end))

    return function(resetPos)
        desiredPos = resetPos
        targetFrame.Position = resetPos
    end
end

function Library:CreateWindow(config)
    config = config or {}
    local TitlePrefix    = config.Title or "tomik"
    local TitleAccent    = config.AccentTitle or ".win"
    if TitlePrefix == "sports" and TitleAccent == "club" then
        TitlePrefix = "tomik"
        TitleAccent = ".win"
    end
    local Subtitle       = config.Subtitle or "Urban Reload"
    local FooterTitle    = config.FooterTitle or "Lifetime"
    local FooterSubtitle = config.FooterSubtitle or "Premium: Yes"
    local LogoUrl        = config.LogoIcon or nil
    local WindowSize     = config.Size or UDim2.fromOffset(710, 535)
    if not config.ForceSize and (WindowSize.X.Offset < 650 or WindowSize.X.Offset > 780) then
        WindowSize = UDim2.fromOffset(710, 535)
    end
    Library.ToggleKeybind = config.ToggleKeybind or Library.ToggleKeybind

    if config.ScriptURL then
        Library.ScriptURL = config.ScriptURL
    end

    local Window = {
        Tabs = {},
        ActiveTab = nil,
        AllGroupboxes = {},
        Visible = config.AutoShow ~= false,
        DefaultPosition = UDim2.fromScale(0.5, 0.5),
        DefaultSize = WindowSize
    }

    -- Outer Window Container (Crisp 1px border, NO background shadow)
    local MainFrame = Create("Frame", {
        Name = "MainWindow",
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = Window.DefaultPosition,
        Size = WindowSize,
        BackgroundColor3 = Library.Theme.Background,
        BorderSizePixel = 0,
        Visible = Window.Visible,
        ClipsDescendants = true,
        Parent = ScreenGui
    })
    AddCorner(MainFrame, 7)
    AddStroke(MainFrame, Library.Theme.Inline, 1)
    Window.MainFrame = MainFrame

    -- Smooth Lerped Window Dragging
    local isDragging = false
    local dragStartMouse = Vector2.zero
    local dragStartPos = MainFrame.Position
    local targetPos = MainFrame.Position

    local function BindDrag(handle)
        handle.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                isDragging = true
                dragStartMouse = Vector2.new(input.Position.X, input.Position.Y)
                dragStartPos = MainFrame.Position
                targetPos = MainFrame.Position
                CloseActivePopover()
                ClickAwayCatcher.Visible = false

                input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then
                        isDragging = false
                    end
                end)
            end
        end)
    end

    -- Bottom-Right Resize Grip
    local ResizeHandle = Create("TextButton", {
        Name = "ResizeHandle",
        AnchorPoint = Vector2.new(1, 1),
        Position = UDim2.new(1, -2, 1, -2),
        Size = UDim2.fromOffset(16, 16),
        BackgroundTransparency = 1,
        Text = "",
        AutoButtonColor = false,
        ZIndex = 20,
        Parent = MainFrame
    })
    Create("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.65, 0, 0.65, 0),
        Size = UDim2.fromOffset(8, 1.5),
        Rotation = -45,
        BackgroundColor3 = Color3.fromRGB(55, 55, 55),
        BorderSizePixel = 0,
        Parent = ResizeHandle
    })
    Create("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.80, 0, 0.80, 0),
        Size = UDim2.fromOffset(4, 1.5),
        Rotation = -45,
        BackgroundColor3 = Color3.fromRGB(55, 55, 55),
        BorderSizePixel = 0,
        Parent = ResizeHandle
    })

    local isResizing = false
    local resizeStartMouse = Vector2.zero
    local resizeStartSize = WindowSize
    local targetSize = WindowSize

    ResizeHandle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            isResizing = true
            resizeStartMouse = Vector2.new(input.Position.X, input.Position.Y)
            resizeStartSize = MainFrame.Size
            targetSize = MainFrame.Size
            CloseActivePopover()
            ClickAwayCatcher.Visible = false
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    isResizing = false
                end
            end)
        end
    end)

    TrackConn(UserInputService.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local mPos = Vector2.new(input.Position.X, input.Position.Y)
            if isDragging then
                local delta = mPos - dragStartMouse
                targetPos = UDim2.new(
                    dragStartPos.X.Scale,
                    dragStartPos.X.Offset + delta.X,
                    dragStartPos.Y.Scale,
                    dragStartPos.Y.Offset + delta.Y
                )
            elseif isResizing then
                local delta = mPos - resizeStartMouse

                local alpha = math.clamp((val - min) / math.max(max - min, 0.0001), 0, 1)
                local dur = instant and 0.05 or 0.14
                FastTween(Fill, dur, { Size = UDim2.new(alpha, 0, 1, 0) })
                FastTween(Thumb, dur, {
                    AnchorPoint = Vector2.new(alpha, 0.5),
                    Position = UDim2.new(alpha, 0, 0.5, 0)
                })
                ValueLabel.Text = FormatVal(val)
                if not skipCallback then
                    if info.Callback then task.spawn(info.Callback, val) end
                    for _, cb in ipairs(SliderObj.ChangedCallbacks) do
                        task.spawn(cb, val)
                    end
                end
            end

            local draggingSlider = false
            local function UpdateFromMouse()
                local mousePos = UserInputService:GetMouseLocation()
                local trackX = Track.AbsolutePosition.X
                local trackW = math.max(Track.AbsoluteSize.X, 1)
                local alpha = math.clamp((mousePos.X - trackX) / trackW, 0, 1)
                SetValue(min + (max - min) * alpha, false, true)
            end

            Hitbox.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    draggingSlider = true
                    -- Lock parent ScrollingFrame while dragging so scrolling never steals mouse drag!
                    if ParentScroll and ParentScroll:IsA("ScrollingFrame") then
                        ParentScroll.ScrollingEnabled = false
                    end
                    SearchLeftCol.ScrollingEnabled = false
                    SearchRightCol.ScrollingEnabled = false
                    FastTween(ThumbStroke, 0.12, { Color = Library.Theme.Accent })
                    UpdateFromMouse()
                end
            end)

            TrackConn(UserInputService.InputEnded:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    if draggingSlider then
                        draggingSlider = false
                        if ParentScroll and ParentScroll:IsA("ScrollingFrame") then
                            ParentScroll.ScrollingEnabled = true
                        end
                        SearchLeftCol.ScrollingEnabled = true
                        SearchRightCol.ScrollingEnabled = true
                        FastTween(ThumbStroke, 0.15, { Color = Color3.fromRGB(64, 64, 64) })
                    end
                end
            end))

            TrackConn(UserInputService.InputChanged:Connect(function(input)
                if draggingSlider and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                    UpdateFromMouse()
                end
            end))

            function SliderObj:SetValue(v) SetValue(v, false, false) end
            function SliderObj:OnChanged(cb) table.insert(SliderObj.ChangedCallbacks, cb) end

                TextBox:GetPropertyChangedSignal("Text"):Connect(function()
                    InputObj.Value = TextBox.Text
                    if info.Callback then task.spawn(info.Callback, TextBox.Text) end
                    for _, cb in ipairs(InputObj.ChangedCallbacks) do
                        task.spawn(cb, TextBox.Text)
                    end
                end)
            end

            function InputObj:SetValue(txt)
                TextBox.Text = tostring(txt or "")
                InputObj.Value = TextBox.Text
            end
            function InputObj:OnChanged(cb) table.insert(InputObj.ChangedCallbacks, cb) end

            Options[idx] = InputObj
            table.insert(Groupbox.Items, { Frame = Wrap, SearchText = labelText })
            return InputObj
        end

                t.View.Visible = false

                Tween(t.Label, nil, { TextColor3 = Library.Theme.Unselected })

                SetHeaderIcon(Tab.ActiveSubTab.IconName)
            else
                SetHeaderIcon(tabIconName)
            end

            if string.lower(tabName) == "settings" then
                TopGearIcon:SetColor(Library.Theme.Accent, 0.2)

                TopGearIcon:SetColor(Library.Theme.Unselected, 0.2)

            Tween(TabStroke, nil, { Transparency = 0 })
            Tween(TabLabel, nil, { TextColor3 = Library.Theme.TextColor })
            TabIcon:SetColor(Library.Theme.Accent, Library.TweenSpeed)

                    st.Page.Visible = false
                    Tween(st.Button, nil, { BackgroundTransparency = 1 })
                    Tween(st.Stroke, nil, { Transparency = 1 })
                    Tween(st.Label, nil, { TextColor3 = Library.Theme.Unselected })

                PageFrame.Visible = true
                if Window.ActiveTab == Tab then

    function Window:AddSettingsTab()
        local SettingsTab = Window:AddTab({
            Name = "Settings",
            Subtitle = "Configs, menu options and theming",
            Icon = "settings"
        })

        local ConfigSubTab = SettingsTab:AddSubTab({
            Name = "Config",
            Icon = "settings"
        })

        local ConfigsBox = ConfigSubTab:AddLeftGroupbox({
            Name = "Configs",
            Icon = "folder"
        })

        local SavedConfigs = {}

        local function ScanDiskConfigs()
            local list = {}
            pcall(function()
                if listfiles and isfolder and isfolder(Library.FolderName .. "/configs") then
                    for _, file in ipairs(listfiles(Library.FolderName .. "/configs")) do
                        local name = string.match(file, "([^/\\]+)%.json$")
                        if name then
                            SavedConfigs[name] = readfile(file)
                        end
                    end
                end
            end)
            for k in pairs(SavedConfigs) do
                table.insert(list, k)
            end
            table.sort(list)
            return list
        end

        local ConfigDropdown = ConfigsBox:AddDropdown("ConfigList", {
            Text = "Configs",
            Values = ScanDiskConfigs(),
            Default = "empty"
        })

        ConfigsBox:AddInput("ConfigName", {
            Text = "Config Name",
            Placeholder = "Name...",
            Inline = true
        })

        ConfigsBox:AddButton("Save", function()
            local name = Options.ConfigName and Options.ConfigName.Value or ""
            if name == "" then
                name = (ConfigDropdown.Value ~= "empty" and ConfigDropdown.Value) or "default"
            end
            local json = Library:GetConfig()
            SavedConfigs[name] = json
            pcall(function()
                if writefile then
                    EnsureFolders()
                    writefile(Library.FolderName .. "/configs/" .. name .. ".json", json)
                end
            end)
            local list = ScanDiskConfigs()
            ConfigDropdown:SetValues(list)
            ConfigDropdown:SetValue(name)
            Library:Notify({ Title = "tomik.win", Description = "Saved config: " .. name })
        end)

        ConfigsBox:AddButton("Load", function()
            local chosen = ConfigDropdown.Value
            if chosen and SavedConfigs[chosen] then
                Library:LoadConfig(SavedConfigs[chosen])
                Library:Notify({ Title = "tomik.win", Description = "Loaded config: " .. chosen })
            else
                Library:Notify({ Title = "tomik.win", Description = "No config selected." })
            end
        end)

        ConfigsBox:AddButton("Export Config", function()
            local json = Library:GetConfig()
            if setclipboard then
                pcall(setclipboard, json)
                Library:Notify({ Title = "tomik.win", Description = "Copied config JSON to clipboard!" })
            else
                Library:Notify({ Title = "tomik.win", Description = "Config exported!" })
            end
        end)

        ConfigsBox:AddButton("Import Config", function()
            if getclipboard then
                local ok, clip = pcall(getclipboard)
                if ok and clip and Library:LoadConfig(clip) then
                    Library:Notify({ Title = "tomik.win", Description = "Imported config from clipboard!" })
                    return
                end
            end
            Library:Notify({ Title = "tomik.win", Description = "No valid config JSON in clipboard." })
        end)

        ConfigsBox:AddButton("Delete", function()
            local chosen = ConfigDropdown.Value
            if chosen and SavedConfigs[chosen] then
                SavedConfigs[chosen] = nil
                pcall(function()
                    if delfile and isfile and isfile(Library.FolderName .. "/configs/" .. chosen .. ".json") then
                        delfile(Library.FolderName .. "/configs/" .. chosen .. ".json")
                    end
                end)
                local list = ScanDiskConfigs()
                ConfigDropdown:SetValues(list)
                ConfigDropdown:SetValue("empty")
                Library:Notify({ Title = "tomik.win", Description = "Deleted config: " .. chosen })
            end
        end)

        end)

        local AutoLoadPath = Library.FolderName .. "/autoload.txt"
        local currentAutoLoad = ""
        pcall(function()
            if isfile and isfile(AutoLoadPath) then
                currentAutoLoad = readfile(AutoLoadPath)
            end

            Text = "Auto Load Config",
            Default = (currentAutoLoad ~= ""),
            Callback = function(state)
                pcall(function()
                    if state then
                        local chosen = (ConfigDropdown.Value ~= "empty" and ConfigDropdown.Value) or "default"
                        if not SavedConfigs[chosen] and writefile then
                            EnsureFolders()
                            writefile(Library.FolderName .. "/configs/" .. chosen .. ".json", Library:GetConfig())
                        end
                        if writefile then
                            writefile(AutoLoadPath, chosen)
                        end
                        Library:Notify({ Title = "tomik.win", Description = "Auto-Load set to: " .. chosen })
                    else
                        if delfile and isfile and isfile(AutoLoadPath) then
                            delfile(AutoLoadPath)
                        end
                    end
                end)
            end
        })

        ConfigsBox:AddToggle("AutoExecuteTeleport", {
            Text = "Auto Execute (Lobby -> Game)",
            Default = true,
            Callback = function(state)
                Library.AutoExecuteOnTeleport = state
                if state then
                    Library:Notify({ Title = "tomik.win", Description = "Will auto-execute when teleporting from Lobby to Match!" })
                end
            end
        })

        -- Automatically load the saved Auto-Load config on startup!
        task.defer(function()
            if currentAutoLoad ~= "" and SavedConfigs[currentAutoLoad] then
                ConfigDropdown:SetValue(currentAutoLoad)
                Library:LoadConfig(SavedConfigs[currentAutoLoad])
                Library:Notify({
                    Title = "tomik.win",

                    Duration = 3
                })
            end

            Name = "Menu",
            Icon = "menu"
        })

            Text = "Menu Bind",
            Default = Library.ToggleKeybind
        })
        MenuBindPicker:OnChanged(function(newKey)
            if newKey and newKey ~= Enum.KeyCode.None then

            end
        end)

        MenuBox:AddToggle("KeybindListToggle", {
            Text = "Keybind List",
            Default = false,
            Callback = function(v)
                RefreshKeybindList()
                KeybindListFrame.Visible = v
            end
        })

        MenuBox:AddToggle("WatermarkToggle", {
            Text = "Watermark",
            Default = false,
            Callback = function(v)
                WatermarkFrame.Visible = v
            end
        })

        MenuBox:AddDropdown("WatermarkOptions", {

        end)

            Default = "Top Left",

        })

            Library:Notify({
                Title = "tomik.win",
                Description = "This is a test notification from tomik.win.",
                Duration = 3.5
            })
        end)

        -- Right Groupbox 2: Theming
        local ThemingBox = ConfigSubTab:AddRightGroupbox({
            Name = "Theming",
            Icon = "theming"
        })

        local themeTokens = {
            { "Accent",             Library.Theme.Accent },
            { "Background",         Library.Theme.Background },
            { "Sidebar",            Library.Theme.Sidebar },
            { "Selected",           Library.Theme.Selected },
            { "Section Background", Library.Theme.SectionBackground },
            { "Section Top",        Library.Theme.SectionTop },
            { "Other Backgrounds",  Library.Theme.OtherBackgrounds },
            { "Inline",             Library.Theme.Inline },
            { "Text Color",         Library.Theme.TextColor },
            { "Unselected",         Library.Theme.Unselected },
        }

            local keyName = token[1]
            ThemingBox:AddColorPicker("Theme_" .. keyName, {
                Text = keyName,
                Default = token[2],
                Callback = function(c)
                    if keyName == "Accent" then
                        Library.Theme.Accent = c
                    elseif keyName == "Background" then
                        Library.Theme.Background = c
                        MainFrame.BackgroundColor3 = c
                        TopHeader.BackgroundColor3 = c
                    elseif keyName == "Sidebar" then
                        Library.Theme.Sidebar = c
                        Sidebar.BackgroundColor3 = c
                    end
                end
            })
        end

        return SettingsTab
    end

    -- Hook Teleport Queue (Lobby -> FFA / Match Transition Auto-Re-Execute!)
    if LocalPlayer then
        TrackConn(LocalPlayer.OnTeleport:Connect(function()
            if not Library.AutoExecuteOnTeleport then return end
            -- Save current state into autoload before teleporting so FFA server loads your exact settings!
            pcall(function()
                if writefile then
                    EnsureFolders()
                    writefile(Library.FolderName .. "/configs/__teleport_autosave.json", Library:GetConfig())
                    if not (isfile and isfile(Library.FolderName .. "/autoload.txt")) then
                        writefile(Library.FolderName .. "/autoload.txt", "__teleport_autosave")
                    end
                end
            end)
            local qFunc = queue_on_teleport or (syn and syn.queue_on_teleport) or (fluxus and fluxus.queue_on_teleport) or queueonteleport
            if qFunc and Library.ExecuteScriptSource then
                pcall(qFunc, Library.ExecuteScriptSource)
            elseif qFunc and Library.ScriptURL then
                pcall(qFunc, string.format('loadstring(game:HttpGet("%s"))()', Library.ScriptURL))
            end
        end))
    end

    -- Keyboard Shortcuts: Ctrl+K to focus Search, MenuBind (LeftControl / RightControl) to toggle UI
    TrackConn(UserInputService.InputBegan:Connect(function(input, gpe)
        if input.KeyCode == Enum.KeyCode.K and UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
            if MainFrame.Visible then
                SearchInput:CaptureFocus()
            end
        elseif not gpe and (input.KeyCode == Library.ToggleKeybind or input.KeyCode == Enum.KeyCode.RightControl) then
            Window.Visible = not Window.Visible
            MainFrame.Visible = Window.Visible
            CloseActivePopover()
            ClickAwayCatcher.Visible = false
        end
    end))

    table.insert(Library.Windows, Window)
    return Window
end

-- Set the script URL or source code to auto-run when teleporting from Lobby to FFA
function Library:SetTeleportScript(urlOrLuaCode)
    if string.find(urlOrLuaCode, "^https?://") then
        Library.ScriptURL = urlOrLuaCode
        Library.ExecuteScriptSource = string.format('repeat task.wait() until game:IsLoaded(); loadstring(game:HttpGet("%s"))()', urlOrLuaCode)
    else
        Library.ExecuteScriptSource = urlOrLuaCode
    end
end

-- Save / Load Configuration
function Library:GetConfig()
    local data = { Toggles = {}, Options = {} }
    for k, v in pairs(Toggles) do
        data.Toggles[k] = v.Value
    end
    for k, v in pairs(Options) do
        if k ~= "ConfigList" and k ~= "ConfigName" then
            if type(v.Value) == "Color3" then
                data.Options[k] = { R = v.Value.R, G = v.Value.G, B = v.Value.B, __type = "Color3" }
            elseif typeof(v.Value) == "EnumItem" then
                data.Options[k] = { Name = v.Value.Name, __type = "KeyCode" }
            else
                data.Options[k] = v.Value
            end
        end
    end
    return HttpService:JSONEncode(data)
end

function Library:LoadConfig(jsonString)
    local ok, data = pcall(function()
        return HttpService:JSONDecode(jsonString)
    end)
    if not ok or type(data) ~= "table" then return false end

    for k, val in pairs(data.Toggles or {}) do
        if Toggles[k] and Toggles[k].SetValue then
            Toggles[k]:SetValue(val)
        end
    end
    for k, val in pairs(data.Options or {}) do
        if Options[k] and k ~= "ConfigList" and k ~= "ConfigName" then
            if type(val) == "table" and val.__type == "Color3" and Options[k].SetValueRGB then
                Options[k]:SetValueRGB(Color3.new(val.R, val.G, val.B))
            elseif Options[k].SetValue then
                Options[k]:SetValue(val)
            end
        end
    end
    return true
end

function Library:Unload()
    for _, conn in ipairs(Library.Connections) do
        pcall(function() conn:Disconnect() end)
    end
    if ScreenGui then
        ScreenGui:Destroy()
    end
end

return Library
